def inv_mod(a, m):
    return pow(a % m, -1, m)


def point_add(P, Q):
    if P is None:
        return Q
    if Q is None:
        return P

    x1, y1 = P
    x2, y2 = Q

    if x1 == x2 and (y1 + y2) % p == 0:
        return None

    if P == Q:
        if y1 == 0:
            return None
        lam = (3 * x1 * x1) * inv_mod(2 * y1, p) % p
    else:
        lam = (y2 - y1) * inv_mod(x2 - x1, p) % p

    x3 = (lam * lam - x1 - x2) % p
    y3 = (lam * (x1 - x3) - y1) % p

    return (x3, y3)

    addend = P

    while k:
        if k & 1:

        addend = point_add(addend, addend)
        k >>= 1

    return result
