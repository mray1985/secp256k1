The missing object is now:

wrap_q = floor((r*d - z) / N)

or, more safely in Python:

residue = (r*d - z) % N
wrap_q = ((r*d - z) - residue) // N

Because the Origin Rule is not just seeing (r*d-z) mod 9; it is seeing:

((r*d - z) mod N) mod 9

and those differ by the amount of N that got subtracted:

((r*d - z) mod N) mod 9
=
((r*d - z) mod 9 - wrap_q * (N mod 9)) mod 9

Since:

N mod 9 = 7

the wrap count contributes:

-wrap_q * 7 mod 9

That is the phase correction.

# Goal:
#   Test whether wrap_q mod 9 is the missing object between:
#       D = delta^-1 mod N - delta^-1 mod p
#   and:
#       Omega = classifier(((r*d - z) mod N) % 9)
#
# Core identity:
#   raw = r*d - z
#   residue_N = raw % N
#   wrap_q = (raw - residue_N) // N
#
#   residue_N % 9 == (raw % 9 - wrap_q*(N % 9)) % 9
#
# Since N mod 9 = 7, wrap_q mod 9 controls the phase shift.
# ============================================================

p = 0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEFFFFFC2F
N = 0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141

delta = p - N

# ------------------------------------------------------------
# Delta inverse flip sequence from true53
# ------------------------------------------------------------

def inv_mod(x, m):
    return pow(x % m, -1, m)

def delta_inverse_flips(rounds=10):
    x = delta
    out = []

    for i in range(1, rounds + 1):
        inv_N = inv_mod(x, N)
        inv_p = inv_mod(x, p)
        D = inv_N - inv_p

        out.append({
            "flip": i,
            "D": D,
            "D_abs": abs(D),
            "D_mod9": D % 9,
            "D_abs_mod9": abs(D) % 9,
            "inv_abs_D_mod_N_mod9": inv_mod(abs(D), N) % 9,
            "inv_abs_D_mod_p_mod9": inv_mod(abs(D), p) % 9,
        })

        x = D

    return out

    name = item["name"]

    r = item["r"]

    z = item["z"]

    residue_N = raw % N

        s = item["s"]
        k = item["k"]
        row["ecdsa_ok"] = ((s * k - (z + r * d)) % N == 0)

def find_linear_maps(xs, ys):
    """
    Brute force maps y = a*x + b mod 9.
    Since mod 9 is not a field, brute force all a,b.
    """
    hits = []
    for a in range(9):
        for b in range(9):
            ok = all(((a * x + b) % 9) == y for x, y in zip(xs, ys))
            if ok:
                hits.append((a, b))
    return hits
